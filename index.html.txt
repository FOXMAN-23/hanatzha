            if (points >= 20) return '🥈 כוכב כסף';

            if(index === '') return;

                updatePersonPoints(index, 5);

        // פונקציית עדכון משודרגת שמעדכנת גם את המונה השנתי
        function updatePersonPoints(index, amount) {
            let oldGreen = people[index].green;
            people[index].green += amount;
            if (people[index].green < 0) people[index].green = 0;
            
            let diff = people[index].green - oldGreen;
            people[index].yearlyGreen = (people[index].yearlyGreen || 0) + diff;
            if (people[index].yearlyGreen < 0) people[index].yearlyGreen = 0;

            renderList();
        }

        function updateBullPoints(amount) {
            if (amount === 'all') { bullPoints = 0; } 
            else { bullPoints += amount; }
            if (bullPoints < 0) bullPoints = 0;
            renderList();
        }

        function addPerson() {
            const inputName = document.getElementById('newName').value.trim();
            
            if (inputName !== '') {
                if(people.some(p => p.name === inputName)) {
                    alert('פעיל/ה בשם זה כבר קיים/ת ברשימה!');
                    return;
                }
                
                let randomAvatar = randomEmojis[Math.floor(Math.random() * randomEmojis.length)];

                people.push({ name: inputName, green: 0, yearlyGreen: 0, avatar: randomAvatar });
                closeModal('addModal'); 
                renderList(); 
            }
        }

        function closeMonth() {
            if (people.length === 0 && bullPoints === 0) {
                alert("אין נתונים החודש!");
                return;
            }
            
            let displayEntities = getSortedEntities();
            const maxScore = displayEntities[0].score;
            if (maxScore === 0) {
                alert("אף אחד לא צבר נקודות החודש!");
                return;
            }

            const topWinners = displayEntities.filter(e => e.score === maxScore);
            const isBullWinner = topWinners.some(e => e.isBull);

            if (isBullWinner) {
                document.body.classList.add('bull-win-animation');
                setTimeout(() => {
                    alert("אוי לא! 🐂 השור סיים במקום הראשון!\nהקבוצה צברה יותר מדי נקודות אדומות והפסידה החודש.");
                    const monthName = prompt("הכנס את שם החודש לתיעוד ניצחון השור בהיכל התהילה (לדוגמה: נובמבר):");
                    if (monthName) {
                        winners.unshift({ month: monthName, name: "השור (הפסד)", score: maxScore, isBull: true });
                        renderWinners();
                        resetEverything();
                    }
                    document.body.classList.remove('bull-win-animation');
                }, 1000);
            } else {
                confetti({ particleCount: 200, spread: 100, origin: { y: 0.6 }, zIndex: 9999 });
                
                setTimeout(() => {
                    let winnerName = topWinners.map(p => p.name).join(" ו-");
                    const monthName = prompt(`המנצח/ת עם ${maxScore} נקודות: ${winnerName}!\n\nהכנס את שם החודש לשמירה בהיכל התהילה (לדוגמה: אוקטובר):`);
                    if (monthName) {
                        winners.unshift({ month: monthName, name: winnerName, score: maxScore, isBull: false });
                        renderWinners();
                        resetEverything();
                    }
                }, 1500);
            }
        }

        function resetEverything() {
            if (confirm("החודש נסגר ונשמר בהצלחה!\nהאם לאפס עכשיו את כל הנקודות כדי להתחיל חודש חדש?\n(הטבלה השנתית לעולם לא תתאפס!)")) {
                people.forEach(p => { p.green = 0; });
                bullPoints = 0;
                renderList();
            }
        }

                renderWinners();
            }
        }
